    37→/**
    38→ * Convert any image file to a PNG data URL, resizing if too large.
    39→ * Fixes JPEG/JPG/WebP compatibility + reduces body size for Vercel/VLM.
    40→ */
    41→async function fileToPngDataUrl(file: File, maxSize: number = 1920): Promise<string> {
    42→  return new Promise((resolve, reject) => {
    43→    const img = new Image()
    44→    img.onload = () => {
    45→      let { width, height } = img
    46→      // Resize if larger than maxSize on any side
    47→      if (width > maxSize || height > maxSize) {
    48→        const ratio = Math.min(maxSize / width, maxSize / height)
    49→        width = Math.round(width * ratio)
    50→        height = Math.round(height * ratio)
    51→      }
    52→      const canvas = document.createElement('canvas')
    53→      canvas.width = width
    54→      canvas.height = height
    55→      const ctx = canvas.getContext('2d')!
    56→      ctx.drawImage(img, 0, 0, width, height)
    57→      const dataUrl = canvas.toDataURL('image/png')
    58→      URL.revokeObjectURL(img.src)
    59→      resolve(dataUrl)
    60→    }
    61→    img.onerror = () => {
    62→      URL.revokeObjectURL(img.src)
    63→      reject(new Error('Failed to load image'))
    64→    }
    65→    img.src = URL.createObjectURL(file)
    66→  })
    67→}
